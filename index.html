"BLOQUEAR":"ATIVAR"; ativar.addEventListener("click",async()=>{ if(!exigirAdmin())return; ativar.disabled=true; try{ await updateDoc(doc(db,"usuarios",item.id),{ ativo:dados.ativo!==true }); mensagemAdmin("Acesso atualizado."); await carregarUsuarios(); }catch(error){ mensagemAdmin(erroFirebase(error)); ativar.disabled=false; } }); div.appendChild(ativar); const excluir=document.createElement("button"); excluir.textContent="🗑 EXCLUIR"; excluir.style.background="#e73535"; excluir.style.marginLeft="8px"; excluir.addEventListener("click",async()=>{ if(!exigirAdmin())return; const confirmar=confirm( "Deseja remover o acesso de "+ (dados.nome||item.id)+"?" ); if(!confirmar)return; excluir.disabled=true; try{ await deleteDoc(doc(db,"usuarios",item.id)); mensagemAdmin("Acesso removido com sucesso."); await carregarUsuarios(); }catch(error){ mensagemAdmin(erroFirebase(error)); excluir.disabled=false; } }); div.appendChild(excluir); } lista.appendChild(div); }); }catch(error){ lista.textContent=""; mensagemAdmin(erroFirebase(error)); } }
